Vendor Threat Radar – verification

Open the Vendor Threat Radar in this tab, then run the checks below in the browser console (F12).

// Paste and run in console on vendor-threat-radar.html:

(function() {
  var out = [];
  function ok(condition, msg) { out.push((condition ? 'PASS' : 'FAIL') + ': ' + msg); }
  function note(msg) { out.push('NOTE: ' + msg); }

  // 1. Mode selector DOM
  var radios = document.querySelectorAll('input[name="radarMode"]');
  ok(radios.length === 3, 'Three mode radios (baseline, real, hybrid)');
  ok(document.getElementById('radarModeBaseline') && document.getElementById('radarModeReal') && document.getElementById('radarModeHybrid'), 'Mode radio IDs present');

  // 2. Mode selector interactive
  var fieldset = document.querySelector('.mode-fieldset');
  var style = fieldset ? getComputedStyle(fieldset) : null;
  ok(!style || style.pointerEvents !== 'none', 'Mode fieldset has pointer-events (not none)');

  // 3. CSV Import
  var csvInput = document.getElementById('csvImport');
  ok(!!csvInput && csvInput.type === 'file', 'File input #csvImport exists');
  ok(typeof window.importVendors === 'function', 'importVendors on window');
  ok(typeof window.downloadVendorCsvTemplate === 'function', 'downloadVendorCsvTemplate on window');

  // 4. Add Vendor
  ok(typeof window.openAddVendor === 'function', 'openAddVendor on window');

  // 5. Template link/button
  var templateBtn = document.getElementById('csvTemplateLink');
  ok(!!templateBtn, 'Download template control (csvTemplateLink) present');

  // 6. Details open by default
  var details = document.querySelector('.csv-format-details');
  ok(!!details, 'CSV format details element present');
  note('details.open = ' + (details ? details.hasAttribute('open') : 'N/A'));

  console.log(out.join('\n'));
  return out;
})();

Expected: All lines should show PASS. If any show FAIL, the radar page has a bug.
